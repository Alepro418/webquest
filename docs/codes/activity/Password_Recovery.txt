@startuml
title Actividad: Recuperación con 5 Preguntas de Seguridad

start

:Usuario selecciona "¿Olvidaste tu contraseña?";
:Sistema solicita nombre_usuario;

:Usuario ingresa nombre_usuario;
:Sistema valida existencia;

if (¿Usuario existe?) then (Sí)
  :Inicializa contador = 1;
  :Inicializa aciertos = 0;
  
  repeat
    :Sistema obtiene pregunta_{contador};
    :Muestra pregunta al usuario;
    :Usuario ingresa respuesta;
    
    :Sistema genera hash de respuesta;
    :Compara con hash almacenado;
    
    if (¿Coincide?) then (Sí)
      :aciertos = aciertos + 1;
    else (No)
      :intentos_fallidos = intentos_fallidos + 1;
      
      if (intentos_fallidos >= 3) then
        :Bloquea proceso por 1 hora;
        :Registra IP y timestamp;
        :Notifica al representante;
        :Muestra mensaje de seguridad;
        stop
      endif
      
      :Muestra: "Respuesta incorrecta";
      :contador = contador - 1; "Reintentar misma pregunta"
    endif
    
    :contador = contador + 1;
    
  repeat while (contador <= 5)
  
  if (aciertos = 5) then
    :Permite establecer nueva contraseña;
    
    repeat
      :Usuario ingresa nueva contraseña;
      :Usuario confirma contraseña;
      
      if (¿Coinciden?) then (No)
        :Muestra error de confirmación;
        backward
      endif
      
      :Sistema valida fortaleza:
      - Mínimo 8 caracteres
      - Al menos 1 mayúscula
      - Al menos 1 número
      - Al menos 1 carácter especial;
      
      if (¿Cumple?) then (No)
        :Muestra requisitos no cumplidos;
        backward
      endif
      
    repeat while (¿Contraseña válida?) is (No)
    
    :Sistema actualiza hash en BD;
    :Registra cambio de contraseña;
    :Envía notificación de cambio;
    :Muestra mensaje de éxito;
    :Redirige a login;
  endif
  
else (No)
  :Muestra: "Usuario no encontrado";
  :Ofrece registrarse;
endif

stop
@enduml
