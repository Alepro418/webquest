@startuml
title Secuencia: Recuperación de Contraseña con 5 Preguntas

actor Estudiante as E
participant "Vista Recuperación" as V
participant "AuthController" as C
participant "UsuarioModel" as UM
participant "PreguntaSeguridadModel" as PM
database "Base de Datos" as DB

E -> V: Selecciona "¿Olvidaste tu contraseña?"
V -> C: mostrarFormularioUsuario()
C --> V: Formulario para nombre_usuario

E -> V: Ingresa nombre_usuario
V -> C: verificarUsuario(usuario)
C -> UM: buscarPorUsuario(usuario)
UM -> DB: SELECT * FROM usuarios
DB --> UM: WHERE nombre_usuario = usuario

alt Usuario existe
    UM --> C: Usuario encontrado
    C -> PM: obtenerPreguntas(id_usuario)
    PM -> DB: SELECT * FROM preguntas_seguridad
    DB --> PM: WHERE id_usuario = id
    PM --> C: Array con 5 preguntas
    C --> V: Muestra formulario con preguntas
    
    loop Pregunta 1 a 5
        E -> V: Responde pregunta
        V -> C: verificarRespuesta(numero, respuesta)
        C -> PM: validarRespuesta(id_usuario, numero, respuesta)
        PM -> PM: generarHash(respuesta)
        PM -> DB: SELECT respuesta_hash
        DB --> PM: hash_almacenado
        
        alt Respuesta correcta
            PM --> C: Válida
            C -> C: incrementar contador aciertos
        else Respuesta incorrecta
            PM --> C: Inválida
            C -> C: incrementar intentos_fallidos
            
            alt Intentos fallidos >= 3
                C -> UM: bloquearProceso(id_usuario)
                UM -> DB: UPDATE usuarios
                C --> V: Mostrar mensaje de bloqueo
                V --> E: "Demasiados intentos. Espere 30 min"
                return
            else
                C --> V: Respuesta incorrecta
                V --> E: "Respuesta incorrecta, intente otra vez"
                ' No hay backward, el bucle sigue a la siguiente pregunta
            end
        end
    end
    
    alt 5 respuestas correctas
        C --> V: Mostrar formulario nueva contraseña
        E -> V: Ingresa nueva contraseña y confirmación
        V -> C: actualizarPassword(usuario, nueva_password)
        C -> UM: cambiarPassword(id_usuario, hash)
        UM -> DB: UPDATE usuarios SET password = hash
        DB --> UM: Confirmación
        UM --> C: Contraseña actualizada
        C --> V: Proceso exitoso
        V --> E: "Contraseña actualizada. Redirigiendo a login"
    end
    
else Usuario no existe
    UM --> C: Usuario no encontrado
    C --> V: Mostrar error
    V --> E: "Usuario no registrado"
end

@enduml
