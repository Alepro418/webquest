@startuml
title Secuencia: Docente - Agregar Recurso Multimedia a Asignatura

actor Docente as D
participant "Vista Recursos" as V
participant "DocenteController" as C
participant "RecursoModel" as RM
participant "FileHelper" as FH
participant "AsignaturaModel" as AM
database "Base de Datos" as DB
participant "Sistema Archivos" as FS

D -> V: Accede a detalle de asignatura
V -> C: showAsignatura(id_asignatura)
C -> AM: obtenerPorId(id_asignatura)
AM -> DB: SELECT * FROM asignaturas
DB --> AM: datos_asignatura
AM --> C: asignatura
C -> RM: obtenerPorAsignatura(id_asignatura)
RM -> DB: SELECT * FROM recursos
DB --> RM: WHERE id_asignatura = id
RM --> C: lista_recursos
C --> V: Muestra asignatura con recursos
V --> D: Vista de asignatura y recursos

D -> V: Hace clic en "Agregar Recurso"
V -> C: showAddResourceForm(id_asignatura)
C --> V: Formulario según tipo
V --> D: Selecciona tipo de recurso

alt Tipo = Enlace
    D -> V: Ingresa URL y título
    V -> C: addLink(id_asignatura, datos)
    C -> C: validarURL(url)
    
    alt URL válida
        C -> RM: nuevoRecurso()
        RM -> RM: setTipo('enlace')
        RM -> RM: setUrl(url)
        RM -> RM: setTitulo(titulo)
        RM -> RM: setIdAsignatura(id_asignatura)
        RM -> DB: INSERT INTO recursos
        DB --> RM: Confirmación
        RM --> C: Recurso creado
        C --> V: Redirigir con éxito
        V --> D: "Enlace agregado correctamente"
    else URL inválida
        C --> V: Error de URL
        V --> D: "URL no válida"
    end
    
else Tipo = Documento
    D -> V: Selecciona archivo local
    V -> FH: subirArchivo(archivo)
    FH -> FH: validarTamaño()
    FH -> FH: validarFormato()
    
    alt Archivo válido
        FH -> FS: guardarArchivo(archivo, '/public/teaching/documents/')
        FS --> FH: ruta_archivo
        FH --> C: ruta_archivo
        
        C -> RM: nuevoRecurso()
        RM -> RM: setTipo('documento')
        RM -> RM: setUrl(ruta_archivo)
        RM -> RM: setTitulo(archivo.nombre)
        RM -> RM: setTamaño(archivo.tamaño)
        RM -> RM: setIdAsignatura(id_asignatura)
        RM -> DB: INSERT INTO recursos
        DB --> RM: Confirmación
        RM --> C: Recurso creado
        C --> V: Redirigir con éxito
        V --> D: "Documento subido correctamente"
    else Archivo inválido
        FH --> C: Error
        C --> V: Mostrar error
        V --> D: "Archivo no válido (máx 10MB)"
    end
    
else Tipo = Video
    D -> V: Ingresa URL de YouTube/Vimeo
    V -> C: addVideo(id_asignatura, url)
    C -> C: extraerIDVideo(url)
    C -> C: obtenerMetadatos(id_video)
    
    alt Video válido
        C -> RM: nuevoRecurso()
        RM -> RM: setTipo('video')
        RM -> RM: setUrl(url_embed)
        RM -> RM: setTitulo(titulo_video)
        RM -> RM: setThumbnail(thumbnail_url)
        RM -> RM: setIdAsignatura(id_asignatura)
        RM -> DB: INSERT INTO recursos
        DB --> RM: Confirmación
        RM --> C: Recurso creado
        C --> V: Redirigir con éxito
        V --> D: "Video agregado correctamente"
    else Video inválido
        C --> V: Error
        V --> D: "URL de video no válida"
    end
end

@enduml
