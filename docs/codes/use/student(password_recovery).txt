@startuml
title Diagrama de Casos de Uso - Recuperación de Acceso con Preguntas de Seguridad

left to right direction
actor "Estudiante" as E

rectangle "Sistema de Recuperación de Acceso" {
  usecase "Solicitar recuperación de contraseña" as UC1
  usecase "Verificar existencia de usuario" as UC2
  usecase "Responder preguntas de seguridad" as UC3
  usecase "Validar respuesta (hash)" as UC4
  usecase "Bloquear intentos por 30 min" as UC5
  usecase "Notificar al representante" as UC6
  usecase "Registrar intento fallido" as UC7
  usecase "Establecer nueva contraseña" as UC8
  usecase "Confirmar nueva contraseña" as UC9
  usecase "Validar fortaleza de contraseña" as UC10
  usecase "Actualizar contraseña en BD" as UC11
  usecase "Notificar cambio exitoso" as UC12
  usecase "Redirigir a login" as UC13
  usecase "Ofrecer registro (si no existe usuario)" as UC14
}

E --> UC1
E --> UC3
E --> UC8
E --> UC9
E --> UC14

' Relaciones de inclusión
UC1 ..> UC2 : <<include>>\n(verificar usuario)
UC3 ..> UC4 : <<include>>\n(validar cada respuesta)
UC8 ..> UC9 : <<include>>\n(confirmar contraseña)
UC8 ..> UC10 : <<include>>\n(validar fortaleza)
UC8 ..> UC11 : <<include>>\n(actualizar en BD)
UC11 ..> UC12 : <<include>>\n(notificar cambio)
UC12 ..> UC13 : <<include>>\n(redirigir a login)

' Relaciones de extensión (excepciones)
UC4 ..> UC5 : <<extend>>\n(si 3 fallos)
UC5 ..> UC6 : <<extend>>\n(notificación opcional)
UC5 ..> UC7 : <<include>>\n(registrar intento)
UC2 ..> UC14 : <<extend>>\n(si usuario no existe)

note right of UC4
  Genera hash de la respuesta
  y compara con hash almacenado
end note

note bottom of UC5
  Bloquea intentos por 30 minutos
  si se superan 3 fallos
end note

note right of UC10
  Valida mínimo 8 caracteres,
  mayúscula, número y especial
end note

note bottom of UC14
  Ofrece opción de registro
  si el usuario no existe
end note
@enduml
