left to right direction
actor "Representante (Padre/Tutor)" as R
actor "Estudiante" as E

rectangle "Sistema de Registro" {
  usecase "Registrar estudiante" as UC1
  usecase "Ingresar datos del representante" as UC2
  usecase "Validar datos del representante" as UC3
  usecase "Ingresar datos del estudiante" as UC4
  usecase "Validar edad (educación primaria)" as UC5
  usecase "Elegir nombre de usuario" as UC6
  usecase "Verificar disponibilidad de usuario" as UC7
  usecase "Crear contraseña" as UC8
  usecase "Confirmar contraseña" as UC9
  usecase "Validar fortaleza de contraseña" as UC10
  usecase "Configurar preguntas de seguridad" as UC11
  usecase "Seleccionar preguntas predefinidas" as UC12
  usecase "Almacenar respuestas (hash)" as UC13
  usecase "Avalar registro (firma digital/términos)" as UC14
  usecase "Generar comprobante de registro" as UC15
  usecase "Enviar notificación (SMS/WhatsApp)" as UC16
  usecase "Activar cuenta del estudiante" as UC17
  usecase "Mostrar tutorial de uso" as UC18
  usecase "Cancelar registro" as UC19
  usecase "Corregir datos" as UC20
}

E --> UC6
E --> UC8

UC8 ..> UC10 : <<include>>

UC11 ..> UC13 : <<include>>
UC1 ..> UC14 : <<include>>
UC14 ..> UC15 : <<include>>
UC14 ..> UC17 : <<include>>

' Relaciones de extensión (opcionales o alternativas)
UC14 ..> UC16 : <<extend>> (notificación opcional)
UC14 ..> UC19 : <<extend>> (si no confirma)
UC19 ..> UC20 : <<include>> (para corregir datos)

note right of UC5
  Valida que el estudiante
  esté en educación primaria
end note

note right of UC7
  Verifica disponibilidad del
  nombre de usuario elegido.
  Si no disponible, sugiere alternativas.
end note

note right of UC10
  Mínimo 8 caracteres,
  mayúscula, número y especial
end note

note right of UC11
  Configura 5 preguntas de seguridad
  de una lista predefinida
end note

note bottom of UC14
  Representante revisa resumen,
  confirma con firma digital/términos
end note

note right of UC16
  Envía SMS/WhatsApp
  con confirmación de registro
end note

note bottom of UC19
  Permite cancelar el registro
  si el representante no confirma
end note
@enduml
